# The player's choices for the runner (JEV|MODE, JEV|PERIOD), at most once a day: on_startup (new games) and every
# weekly and monthly update, since on_startup does not run for a loaded save. Also gives saves from JevAI 0.2 (no
# period) their old pace, and asks the start-of-game questions in a campaign that has never answered them (jevai.1
# fires only once).
jev_announce = {
	if = {
		limit = { NOT = { has_global_flag = jevai_announced } }
		set_global_flag = { flag = jevai_announced value = 1 days = 1 }
		if = {
			limit = {
				NOT = { has_global_flag = jevai_period_week }
				NOT = { has_global_flag = jevai_period_month }
				NOT = { has_global_flag = jevai_period_quarter }
			}
			if = { limit = { has_global_flag = jevai_mode_majors } set_global_flag = jevai_period_week }
			if = { limit = { has_global_flag = jevai_mode_all } set_global_flag = jevai_period_month }
		}
		if = { limit = { has_global_flag = jevai_mode_all } log = "JEV|MODE|all" }
		if = { limit = { has_global_flag = jevai_mode_majors } log = "JEV|MODE|majors" }
		if = { limit = { has_global_flag = jevai_mode_off } log = "JEV|MODE|off" }
		if = { limit = { has_global_flag = jevai_period_week } log = "JEV|PERIOD|week" }
		if = { limit = { has_global_flag = jevai_period_month } log = "JEV|PERIOD|month" }
		if = { limit = { has_global_flag = jevai_period_quarter } log = "JEV|PERIOD|quarter" }
		if = {
			limit = {
				NOT = { has_global_flag = jevai_mode_majors }
				NOT = { has_global_flag = jevai_mode_all }
				NOT = { has_global_flag = jevai_mode_off }
			}
			every_country = {
				limit = { is_ai = no }
				country_event = { id = jevai.1 days = 1 }
			}
		}
	}
}

# What the runner (jevai.exe, runtime/runner.py log_records) reads about a country, logged to game.log:
# T = stability, war support, strength ratios, posture in force; S = the numbers of the state text; G = ideology group
# key (language independent); H = human-played; MAJOR; N/E/A = one line per neighbour / enemy / ally.
# Called monthly for every country and, with the weekly period, weekly for the countries in range (on_actions/jevai.txt).
jev_log_state = {
	log = "JEV|T|[ROOT.GetTag]|ideo=[ROOT.GetRulingIdeology]|fac=[ROOT.GetFactionName]|stab=[?ROOT.stability|2]|ws=[?ROOT.has_war_support|2]|str=[?ROOT.enemies_strength_ratio|2]|astr=[?ROOT.alliance_strength_ratio|2]|surr=[?ROOT.surrender_progress|2]|pp=[?ROOT.political_power|0]|mp=[?ROOT.manpower_k|0]|cas=[?ROOT.casualties_k|0]|subj=[?ROOT.num_subjects|0]|pos=[?ROOT.jev_posture|0]"
	log = "JEV|S|[ROOT.GetTag]|name=[ROOT.GetName]|mil=[?ROOT.num_of_military_factories|0]|civ=[?ROOT.num_of_civilian_factories|0]|nav=[?ROOT.num_of_naval_factories|0]|st=[?ROOT.num_of_controlled_states|0]|div=[?ROOT.num_divisions|0]|mpmax=[?ROOT.max_manpower_k|0]"
	if = { limit = { has_government = fascism } log = "JEV|G|[ROOT.GetTag]|fascism" }
	if = { limit = { has_government = communism } log = "JEV|G|[ROOT.GetTag]|communism" }
	if = { limit = { has_government = democratic } log = "JEV|G|[ROOT.GetTag]|democratic" }
	if = { limit = { has_government = neutrality } log = "JEV|G|[ROOT.GetTag]|neutrality" }
	if = { limit = { is_ai = no } log = "JEV|H|[ROOT.GetTag]" }
	if = { limit = { is_major = yes } log = "JEV|MAJOR|[ROOT.GetTag]" }
	every_neighbor_country = { log = "JEV|N|[ROOT.GetTag]|[THIS.GetTag]" }
	every_enemy_country = { log = "JEV|E|[ROOT.GetTag]|[THIS.GetTag]" }
	every_allied_country = { log = "JEV|A|[ROOT.GetTag]|[THIS.GetTag]" }
}
