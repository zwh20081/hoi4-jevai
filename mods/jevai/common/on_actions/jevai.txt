on_actions = {
	on_startup = {
		effect = {
			log = "JEV|START|[GetDateText]"
			# the runner reads the chosen mode from game.log (the event logs it when answered; re-logged on each load)
			if = { limit = { has_global_flag = jevai_mode_all } log = "JEV|MODE|all" }
			if = { limit = { has_global_flag = jevai_mode_majors } log = "JEV|MODE|majors" }
			if = { limit = { has_global_flag = jevai_mode_off } log = "JEV|MODE|off" }
			# ask each human player once which AI countries JevAI should steer (events/jevai.txt)
			if = {
				limit = { NOT = { has_global_flag = jevai_mode_majors has_global_flag = jevai_mode_all has_global_flag = jevai_mode_off } }
				every_country = {
					limit = { is_ai = no }
					country_event = { id = jevai.1 days = 1 }
				}
			}
		}
	}
	# The runner (jevai.exe) rewrites history/units/JEVAI_orders.txt with each AI country's wanted posture (jev_want).
	# on_daily runs once per country (ROOT = the country). The first country each day reloads the orders file for
	# everyone (the timed global flag keeps it to one load per day, whatever tags the mod set has), so a decision takes
	# effect the next day; every country then applies its own order. jev_follow_orders only touches AI countries and
	# only when the wanted posture changed.
	on_daily = {
		effect = {
			if = {
				limit = { NOT = { has_global_flag = jevai_orders_loaded } }
				set_global_flag = { flag = jevai_orders_loaded value = 1 days = 1 }
				load_oob = "JEVAI_orders"
			}
			if = {
				limit = { has_variable = jev_want }
				jev_follow_orders = yes
			}
		}
	}
	# The runner's view of each country (scripted_effects/jevai_state.txt): monthly for every country and, in the
	# "major powers only" mode, weekly for major powers, which the runner then re-decides every week.
	on_weekly = {
		effect = {
			if = {
				limit = { has_global_flag = jevai_mode_majors is_major = yes num_of_controlled_states > 0 }
				jev_log_state = yes
			}
		}
	}
	# P (cumulative production) is only for the training data (it lands in -dump_history's `logs`, like T).
	on_monthly = {
		effect = {
			if = {
				limit = { num_of_controlled_states > 0 }
				jev_log_state = yes
				log = "JEV|P|[ROOT.GetTag]|inf=[?ROOT.total_equipment_produced_infantry|0]|art=[?ROOT.total_equipment_produced_artillery|0]|at=[?ROOT.total_equipment_produced_anti_tank|0]|aa=[?ROOT.total_equipment_produced_anti_air|0]|arm=[?ROOT.total_equipment_produced_armor|0]|mot=[?ROOT.total_equipment_produced_motorized|0]|fig=[?ROOT.total_equipment_produced_fighter|0]|cas=[?ROOT.total_equipment_produced_cas|0]|tac=[?ROOT.total_equipment_produced_tactical_bomber|0]|nav=[?ROOT.total_equipment_produced_naval_bomber|0]|scr=[?ROOT.total_equipment_produced_screen_ship|0]|cap=[?ROOT.total_equipment_produced_capital_ship|0]|sub=[?ROOT.total_equipment_produced_submarine|0]|bld=[?ROOT.total_constructed_civilian_factory|0]/[?ROOT.total_constructed_military_factory|0]/[?ROOT.total_constructed_dockyard|0]"
			}
		}
	}
}
